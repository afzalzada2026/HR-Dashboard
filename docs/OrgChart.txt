const CARD = { w: 184, h: 100 };
const COMPACT = { w: 152, h: 78 };
const initials = (e: Employee) => `${e.firstName?.[0] ?? e.fullName?.[0] ?? ""}${e.lastName?.[0] ?? ""}`.toUpperCase();
const uniq = (arr: string[]) => Array.from(new Set(arr.filter(Boolean))).sort();
const empKey = (e: Employee) => `${e.employeeNo}|${e.email}|${e.fullName}`;
const displayLevel = (label?: string) => (label || "—").replace(/^L/i, "");
const positionStatus = (e: Employee) => {
  const all = `${e.fullName} ${e.title} ${e.remarks}`.toLowerCase();
  const vacant = !e.fullName || /\b(vacant|vacancy|unfilled|tbd|to be hired)\b/.test(all) || /^(unknown|vacant)$/i.test(e.fullName.trim());
  const temporary = !vacant && /\b(temp|temporary|contract|contractor|acting|interim)\b/.test(all);
  return { vacant, temporary };
};
/** Suppresses the click that follows a drag-to-pan gesture (module-level: one canvas interacts at a time). */
const dragGuard = { moved: false };

function useOrgModel(data: Employee[], pref: SeniorityDirection, headOverrides: Record<string, string>, reportingOverrides: Record<string, string>) {
  const build = useMemo(() => buildOrgTree(data, pref, headOverrides, reportingOverrides), [data, pref, headOverrides, reportingOverrides]);
  const levels = useMemo(() => analyzeLevels(data, build.direction), [data, build.direction]);
  const lidx = useMemo(() => levelIndex(levels), [levels]);
  const detection = useMemo(() => resolveDirection(pref, data), [pref, data]);
  const levelOf = (e: Employee): LevelInfo | undefined => lidx.get(levelKey(e));
  return { build, levels, levelOf, detection };
}

interface Ctx {
  levelOf: (e: Employee) => LevelInfo | undefined;
  matches: Set<string>;
  dim: Set<string> | null;
  compact: boolean;
  interactive: boolean;
  isOpen: (id: string) => boolean;
  toggle: (id: string) => void;
  select: (e: Employee) => void;
}

/* ---------------- canvas ---------------- */
function OrgCanvas({ layout, levels, ctx, zoom, maxHeight, containerRef, interactive, scopeLabel }: {
  layout: OrgLayout; levels: LevelInfo[]; ctx: Ctx; zoom: number; maxHeight?: number; containerRef: React.RefObject<HTMLDivElement | null>; interactive: boolean; scopeLabel: string;
}) {
  const drag = useRef<{ x: number; y: number; sl: number; st: number; moved: boolean } | null>(null);
  const onDown = (e: React.PointerEvent<HTMLDivElement>) => {
    if (!interactive || e.button !== 0) return;
    const el = containerRef.current; if (!el) return;
    drag.current = { x: e.clientX, y: e.clientY, sl: el.scrollLeft, st: el.scrollTop, moved: false };
  };
  const onMove = (e: React.PointerEvent<HTMLDivElement>) => {
    const d = drag.current, el = containerRef.current; if (!d || !el) return;
    const dx = e.clientX - d.x, dy = e.clientY - d.y;
    if (!d.moved && Math.abs(dx) + Math.abs(dy) < 5) return;
    d.moved = true; el.scrollLeft = d.sl - dx; el.scrollTop = d.st - dy;
  };
  const onUp = () => {
    const moved = !!drag.current?.moved; drag.current = null;
    if (moved) { dragGuard.moved = true; window.setTimeout(() => { dragGuard.moved = false; }, 0); }
  };
  return (
    <div ref={containerRef} onPointerDown={onDown} onPointerMove={onMove} onPointerUp={onUp} onPointerLeave={onUp}
      className={`org-canvas scroll-thin relative rounded-xl border border-app bg-white/40 ${interactive ? "cursor-grab select-none overflow-auto active:cursor-grabbing" : "overflow-hidden"}`}
      style={{ maxHeight }}>
      <div style={{ width: layout.width * zoom, height: layout.height * zoom, position: "relative" }}>
        <div style={{ width: layout.width, height: layout.height, transform: `scale(${zoom})`, transformOrigin: "0 0", position: "absolute", left: 0, top: 0 }}>
          <div className="absolute bottom-0 top-0 bg-amber-200/75" style={{ left: 12, width: 48 }} />
          {layout.layers.map((L) => {
            const info = levels[L.actualPosition];
            const start = Math.max(0, L.offset - layout.gapMain / 2);
            const size = L.size + layout.gapMain;
            return (
              <div key={L.index} className="absolute left-0" style={{ top: start, width: layout.width, height: size, background: L.index % 2 ? "rgba(13,71,161,0.025)" : "transparent", borderTop: L.index > 0 ? "2px solid #F8B900" : undefined }}>
                <div className="absolute flex flex-col items-center justify-center text-center" style={{ left: 12, top: 0, width: 48, height: size }}>
                  <span className="text-xs font-extrabold text-slate-900">{displayLevel(info?.label)}{L.continuation ? " cont." : ""}</span>
                  <span className="mt-0.5 text-[8px] font-semibold leading-tight text-slate-700">{L.count}</span>
                </div>
                <div className="absolute left-[66px] top-1 rounded-md bg-white/80 px-1.5 py-0.5 text-[9px] font-bold text-slate-600 shadow-sm">{info?.band ?? "Unspecified"}</div>
              </div>
            );
          })}
          <svg width={layout.width} height={layout.height} style={{ position: "absolute", left: 0, top: 0, pointerEvents: "none", overflow: "visible" }}>
            {layout.edges.map((e) => <path key={e.id} d={e.d} fill="none" stroke={e.inferred ? "#F59E0B" : "#1F2937"} strokeWidth={e.inferred ? 1.5 : 1.2} strokeDasharray={e.inferred ? "5 4" : undefined} strokeLinejoin="round" strokeLinecap="round" />)}
          </svg>
          {layout.nodes.map((ln) => {
            if (ln.kind === "node") return <NodeCard key={ln.id} ln={ln} ctx={ctx} />;
            const info = levels[layout.layers[ln.layer]?.actualPosition ?? -1];
            return <TeamCard key={ln.id} ln={ln} ctx={ctx} levelLabel={displayLevel(info?.label)} color={info?.color ?? "#94A3B8"} />;
          })}
          {layout.nodes.length === 0 && <div className="p-10 text-center text-xs text-muted-app">No employees in this scope.</div>}
        </div>
      </div>
    </div>
  );
}

export function LevelPyramid({ levels, total }: { levels: LevelInfo[]; total: number }) {
  const max = Math.max(1, ...levels.map((l) => l.count));
  return (
    <div className="space-y-1">
      {levels.map((l) => (
        <div key={l.label} className="grid grid-cols-[52px_1fr_150px] items-center gap-2 text-[11px]">
          <span className="rounded-md px-1.5 py-0.5 text-center font-extrabold text-white" style={{ background: l.color }}>{displayLevel(l.label)}</span>
          <div className="flex justify-center"><div className="h-5 rounded-md" style={{ width: `${Math.max(3, (l.count / max) * 100)}%`, background: `linear-gradient(90deg, ${l.color}, ${l.color}bb)` }} /></div>
          <span className="truncate text-muted-app"><b className="text-app">{l.count}</b> · {((l.count / Math.max(1, total)) * 100).toFixed(1)}% · {l.band}</span>
        </div>
      ))}
    </div>
  );
}

function LevelLanes({ nodes, levels, ctx }: { nodes: OrgNode[]; levels: LevelInfo[]; ctx: Ctx }) {
  const byLevel = useMemo(() => {
    const m = new Map<string, OrgNode[]>();
    nodes.forEach((n) => { const k = levelKey(n.emp); if (!m.has(k)) m.set(k, []); m.get(k)!.push(n); });
    m.forEach((arr) => arr.sort((a, b) => b.total - a.total || a.emp.fullName.localeCompare(b.emp.fullName)));
    return m;
  }, [nodes]);
  return (
    <div className="space-y-2">
      {levels.map((l) => {
        const arr = byLevel.get(l.label) ?? [];
        const shown = arr.slice(0, 48);
        return (
          <div key={l.label} className="grid gap-2 rounded-xl border border-app bg-white/30 p-2 lg:grid-cols-[190px_1fr]">
            <div className="flex items-center gap-2 lg:border-r lg:border-app lg:pr-2">
              <span className="rounded-md px-2 py-1 text-xs font-extrabold text-white" style={{ background: l.color }}>{displayLevel(l.label)}</span>
              <div className="min-w-0"><div className="truncate text-xs font-extrabold text-app">{l.band}</div><div className="text-[10px] text-muted-app">{arr.length} · avg {l.avgAge.toFixed(0)} yrs · {l.female} F</div></div>
            </div>
            <div className="flex flex-wrap gap-1">
              {shown.map((n) => (
                <button key={n.id} data-org-node={n.id} onClick={() => ctx.select(n.emp)} title={`${n.emp.title} · ${n.emp.department}${n.children.length ? ` · ${n.children.length} direct reports` : ""}`} className={`flex items-center gap-1 rounded-lg border border-app bg-white/50 px-1.5 py-1 text-left hover:border-[#00A8FF] ${ctx.matches.has(n.id) ? "ring-2 ring-[#00A8FF]" : ""} ${ctx.dim && !ctx.dim.has(empKey(n.emp)) ? "opacity-35" : ""}`}>
                  <span className="grid h-5 w-5 shrink-0 place-items-center rounded-full text-[8px] font-bold text-white" style={{ background: l.color }}>{initials(n.emp)}</span>
                  <span className="max-w-[120px] truncate text-[10px] font-bold text-app">{n.emp.fullName}</span>
                  {n.children.length > 0 && <span className="rounded bg-black/5 px-1 text-[9px] font-bold text-muted-app">{n.children.length}</span>}
                </button>
              ))}
              {arr.length > shown.length && <span className="self-center px-1 text-[10px] font-semibold text-muted-app">+{arr.length - shown.length} more</span>}
              {arr.length === 0 && <span className="self-center text-[10px] text-muted-app">—</span>}
            </div>
          </div>
        );
      })}
    </div>
  );
}

const layerOfFactory = (levelOf: (e: Employee) => LevelInfo | undefined, fallback: number) => (n: OrgNode) => levelOf(n.emp)?.position ?? fallback;

  const [mode, setMode] = useState<"chart" | "lanes">("chart");
  const [compact, setCompact] = useState(false);
  // ATOMA divisions currently stay below 80 people: show the full division by default. Above 80,
  // fall back to a leadership view; the export toolbar can still include the full hierarchy.
  const initialDivision = filters.divisions.length === 1 ? filters.divisions[0] : "";
  const initialScopeSize = initialDivision ? data.filter((e) => e.division === initialDivision).length : data.length;
  const [expandPos, setExpandPos] = useState(filters.departments.length === 1 || (initialDivision && initialScopeSize <= 80) ? -1 : initialDivision ? 2 : 0);
  const [expanded, setExpanded] = useState<Set<string> | null>(null);
  const [showUnmatched, setShowUnmatched] = useState(false);
  const [showHeadEditor, setShowHeadEditor] = useState(false);
  const [lineEmployee, setLineEmployee] = useState("");
  const [lineManager, setLineManager] = useState("");
  const [exportFull, setExportFull] = useState(true);
  const [exportBusy, setExportBusy] = useState<"" | "vsdx" | "pdf" | "pptx">("");
  const [exportMsg, setExportMsg] = useState("");
  const containerRef = useRef<HTMLDivElement>(null);
  const layoutRef = useRef<OrgLayout | null>(null);

  const divisions = useMemo(() => uniq(data.map((e) => e.division)), [data]);
  const departments = useMemo(() => uniq(data.filter((e) => !division || e.division === division).map((e) => e.department)), [data, division]);
  const roots = useMemo(() => scopeRoots(build, division || undefined, department || undefined), [build, division, department]);
  const displayRoots = roots;
  const inScope = useMemo(() => build.nodes.filter((n) => (!division || n.emp.division === division) && (!department || n.emp.department === department)), [build, division, department]);
  const scopeLevels = useMemo(() => analyzeLevels(inScope.map((n) => n.emp), build.direction), [inScope, build.direction]);
  const scopeLabel = department || division || "ATOMA";

  const dim = useMemo(() => (highlight && highlight.length !== data.length ? new Set(highlight.map(empKey)) : null), [highlight, data]);
  const matches = useMemo(() => {
    const s = q.trim().toLowerCase();
    if (!s) return new Set<string>();
    return new Set(build.nodes.filter((n) => `${n.emp.fullName} ${n.emp.title} ${n.emp.employeeNo} ${n.emp.department} ${n.emp.email}`.toLowerCase().includes(s)).map((n) => n.id));
  }, [q, build]);
  const limitSen = expandPos < 0 ? -Infinity : levels[Math.min(expandPos, Math.max(0, levels.length - 1))]?.sen ?? -Infinity;
  const openSet = useMemo(() => {
    const base = expanded ? new Set(expanded) : new Set(build.nodes.filter((n) => n.sen >= limitSen).map((n) => n.id));
    matches.forEach((id) => ancestorIds(build, id).forEach((a) => base.add(a)));
    displayRoots.forEach((r) => base.add(r.id));
    return base;
  }, [expanded, build, limitSen, matches, displayRoots]);

  const layout = useMemo(() => layoutOrg(displayRoots, {
    layerOf: layerOfFactory(levelOf, levels.length), isOpen: (id) => openSet.has(id), card: compact ? COMPACT : CARD,
  }), [displayRoots, levelOf, levels.length, openSet, compact]);
  useEffect(() => { layoutRef.current = layout; });
  const exportOpen = useMemo(() => exportFull ? new Set(build.nodes.map((n) => n.id).concat(displayRoots.map((r) => r.id))) : openSet, [exportFull, build.nodes, displayRoots, openSet]);
  const exportLayout = useMemo(() => layoutOrg(displayRoots, {
    layerOf: layerOfFactory(levelOf, levels.length), isOpen: (id) => exportOpen.has(id), card: COMPACT,
    groupThreshold: 4, groupMax: 1000,
  }), [displayRoots, levelOf, levels.length, exportOpen]);
  const exportOpts: OrgExportOpts = {
    title: `ATOMA Organogram — ${department || division || "Whole Organization"}`,
    subtitle: `${inScope.length.toLocaleString()} positions · levels ${levels[exportLayout.levelOffset]?.label?.replace(/^L/, "") || "—"} to ${levels[Math.min(levels.length - 1, exportLayout.levelOffset + exportLayout.layers.length - 1)]?.label?.replace(/^L/, "") || "—"} · ${exportFull ? "full hierarchy" : "current expansion"}`,
    generatedBy: useHR.getState().session?.displayName,
    footer: "Made with ♥ by Mohibullah Afzalzada",
  };
  const runExport = async (kind: "svg" | "vsdx" | "pdf" | "pptx" | "print") => {
    setExportMsg("");
    try {
      if (kind === "svg") { downloadOrgSvg(exportLayout, levels, exportOpts); setExportMsg("Editable SVG downloaded — vector quality at any zoom."); }
      if (kind === "vsdx") { setExportBusy("vsdx"); const stats = await downloadOrgVisio(exportLayout, levels, exportOpts); setExportMsg(`Editable Visio VSDX downloaded — ${stats.shapes} native shapes, ${stats.texts} text elements.`); }
      if (kind === "pdf") { setExportBusy("pdf"); await downloadOrgVectorPdf(exportLayout, levels, exportOpts); setExportMsg("Single-page vector PDF downloaded."); }

/* ---------------- print / report variant ---------------- */
export function OrgChartPrint({ data, division, department, expandPos, maxWidth = 1352 }: { data: Employee[]; division?: string; department?: string; expandPos: number; maxWidth?: number }) {
  const pref = useHR((s) => s.seniorityDirection);
  const headOverrides = useHR((s) => s.divisionHeadOverrides);
  const reportingOverrides = useHR((s) => s.reportingLineOverrides);
  const { build, levels, levelOf } = useOrgModel(data, pref, headOverrides, reportingOverrides);
  const roots = useMemo(() => scopeRoots(build, division, department), [build, division, department]);
  const displayRoots = roots;
  const inScope = useMemo(() => build.nodes.filter((n) => (!division || n.emp.division === division) && (!department || n.emp.department === department)), [build, division, department]);
  const limitSen = expandPos < 0 ? -Infinity : levels[Math.min(expandPos, Math.max(0, levels.length - 1))]?.sen ?? -Infinity;
  const open = useMemo(() => { const s = new Set(build.nodes.filter((n) => n.sen >= limitSen).map((n) => n.id)); roots.forEach((r) => s.add(r.id)); return s; }, [build, limitSen, roots]);
  const scopeLabel = department || division || "ATOMA";
  const layout = useMemo(() => layoutOrg(displayRoots, {
    layerOf: layerOfFactory(levelOf, levels.length), isOpen: (id) => open.has(id), card: COMPACT,
  }), [displayRoots, levelOf, levels.length, open]);
  const zoom = Math.max(0.2, Math.min(1, maxWidth / Math.max(1, layout.width)));
  const containerRef = useRef<HTMLDivElement>(null);
  const ctx: Ctx = { levelOf, matches: new Set(), dim: null, compact: true, interactive: false, isOpen: (id) => open.has(id), toggle: () => {}, select: () => {} };
  const order = levels.filter((l) => l.band !== "Unspecified").map((l) => displayLevel(l.label));
  return (
    <div>
      <div className="mb-2 flex flex-wrap items-center gap-2 text-[11px] text-muted-app">
        <span className="font-bold text-app">{department || division || "Whole organization"}</span> · {inScope.length.toLocaleString()} employees · landscape · rows = levels {order.join(" › ")} · expanded to {expandPos < 0 ? "everyone" : displayLevel(levels[expandPos]?.label)} · dashed = inferred line
      </div>
      <OrgCanvas layout={layout} levels={levels} ctx={ctx} zoom={zoom} containerRef={containerRef} interactive={false} scopeLabel={scopeLabel} />
      <div className="mt-2"><LevelLegend levels={levels} /></div>
    </div>
  );
}
